Sub CombineSplitSalesWithTotals()
    Dim wsReport As Worksheet, wsItems As Worksheet
    Dim wsB2B As Worksheet, wsB2C As Worksheet, wsCash As Worksheet
    Dim sht As Worksheet
    Dim lastRowReport As Long, lastRowItems As Long
    Dim outB2B As Long, outB2C As Long, outCash As Long
    Dim i As Long, j As Long
    Dim dictGSTIN As Object, dictPayType As Object
    Dim key As String, headerText As String
    Dim rawDate As String, properDate As Variant
    Dim gstStr As String, discStr As String
    Dim splitData() As String
    
    ' --- DYNAMIC COLUMN VARIABLES ---
    Dim rRow As Long, iRow As Long
    Dim rInvCol As Long, rGSTINCol As Long, rPayCol As Long
    Dim iDateCol As Long, iPartyCol As Long, iInvCol As Long, iItemCol As Long
    Dim iHSNCol As Long, iQtyCol As Long, iPriceCol As Long, iDiscCol As Long
    Dim iGSTCol As Long, iAmtCol As Long

    ' 1. Safely find the Sales sheets
    For Each sht In ThisWorkbook.Sheets
        If InStr(1, sht.Name, "Sale Report", vbTextCompare) > 0 Then Set wsReport = sht
        If InStr(1, sht.Name, "Sale Items", vbTextCompare) > 0 Then Set wsItems = sht
    Next sht

    If wsReport Is Nothing Or wsItems Is Nothing Then
        MsgBox "Could not find 'Sale Report' or 'Sale Items'. Please check your tab names.", vbCritical
        Exit Sub
    End If

    ' 2. Clean recreate the target accounting sheets
    Application.DisplayAlerts = False
    On Error Resume Next
    ThisWorkbook.Sheets("Miracle Registered B2B").Delete
    ThisWorkbook.Sheets("Miracle Unregistered B2C").Delete
    ThisWorkbook.Sheets("Miracle Cash Sales").Delete
    On Error GoTo 0
    Application.DisplayAlerts = True
    
    Set wsB2B = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
    wsB2B.Name = "Miracle Registered B2B"
    
    Set wsB2C = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
    wsB2C.Name = "Miracle Unregistered B2C"
    
    Set wsCash = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
    wsCash.Name = "Miracle Cash Sales"

    ' 3. Set Headers for All 3 Target Sheets
    Dim headers As Variant
    headers = Array("Date", "Party Name", "Party's GSTIN No.", "Invoice No.", "Item Name", _
                    "HSN/SAC", "Quantity", "Price/Unit", "Discount Amount", "Discount %", "GST Amount", "GST %", "Total Amount", "Payment Type")
    
    Dim wsTarget As Variant
    For Each wsTarget In Array(wsB2B, wsB2C, wsCash)
        wsTarget.Range("A1").Resize(1, UBound(headers) + 1).Value = headers
        wsTarget.Range("A1").Resize(1, UBound(headers) + 1).Font.Bold = True
    Next wsTarget
    
    wsB2B.Range("A1:N1").Interior.Color = RGB(200, 230, 201) ' Soft Green
    wsB2C.Range("A1:N1").Interior.Color = RGB(174, 214, 241) ' Soft Blue
    wsCash.Range("A1:N1").Interior.Color = RGB(255, 235, 165) ' Soft Yellow

    ' 4. Find Column Headers Dynamically
    rRow = 1: iRow = 1
    
    For j = 1 To 15
        headerText = Replace(Replace(wsReport.Cells(rRow, j).Text, Chr(10), ""), Chr(13), "")
        If InStr(1, headerText, "Invoice No", vbTextCompare) > 0 Then rInvCol = j
        If InStr(1, headerText, "GSTIN", vbTextCompare) > 0 Then rGSTINCol = j
        If InStr(1, headerText, "Payment Type", vbTextCompare) > 0 Then rPayCol = j
    Next j

    For j = 1 To 20
        headerText = Replace(Replace(wsItems.Cells(iRow, j).Text, Chr(10), ""), Chr(13), "")
        If InStr(1, headerText, "Date", vbTextCompare) > 0 Then iDateCol = j
        If InStr(1, headerText, "Party", vbTextCompare) > 0 Then iPartyCol = j
        If InStr(1, headerText, "Invoice", vbTextCompare) > 0 Then iInvCol = j
        If InStr(1, headerText, "Item Name", vbTextCompare) > 0 Then iItemCol = j
        If InStr(1, headerText, "HSN", vbTextCompare) > 0 Then iHSNCol = j
        If InStr(1, headerText, "Quantit", vbTextCompare) > 0 Then iQtyCol = j
        If InStr(1, headerText, "Price", vbTextCompare) > 0 Then iPriceCol = j
        If InStr(1, headerText, "Discount", vbTextCompare) > 0 Then iDiscCol = j
        If InStr(1, headerText, "GST", vbTextCompare) > 0 And InStr(1, headerText, "GSTIN", vbTextCompare) = 0 Then iGSTCol = j
        If InStr(1, headerText, "Amount", vbTextCompare) > 0 And InStr(1, headerText, "GST", vbTextCompare) = 0 And InStr(1, headerText, "Discount", vbTextCompare) = 0 Then iAmtCol = j
    Next j

    If rInvCol = 0 Or iInvCol = 0 Or rPayCol = 0 Then
        MsgBox "Error: Required headers are missing from your export layout.", vbCritical
        Exit Sub
    End If

    ' 5. Map GSTIN and Payment Type from Sale Report into Dictionaries
    Set dictGSTIN = CreateObject("Scripting.Dictionary")
    Set dictPayType = CreateObject("Scripting.Dictionary")
    lastRowReport = wsReport.Cells(wsReport.Rows.Count, rInvCol).End(xlUp).Row

    Dim tempVal As Variant
    For i = rRow + 1 To lastRowReport
        tempVal = wsReport.Cells(i, rInvCol).Value
        If Not IsError(tempVal) Then
            key = Trim(CStr(tempVal))
            If key <> "" And LCase(key) <> "total" Then
                If rGSTINCol > 0 And Not dictGSTIN.exists(key) Then dictGSTIN.Add key, Trim(wsReport.Cells(i, rGSTINCol).Text)
                If Not dictPayType.exists(key) Then dictPayType.Add key, Trim(wsReport.Cells(i, rPayCol).Text)
            End If
        End If
    Next i

    ' 6. Process Items and Sort Lines to Correct Sheet
    lastRowItems = wsItems.Cells(wsItems.Rows.Count, iInvCol).End(xlUp).Row
    outB2B = 2: outB2C = 2: outCash = 2

    Dim chosenWs As Worksheet, outRow As Long
    Dim currentGSTIN As String, currentPayType As String

    For i = iRow + 1 To lastRowItems
        If wsItems.Rows(i).Hidden = False Then
            tempVal = wsItems.Cells(i, iInvCol).Value
            
            If Not IsError(tempVal) Then
                key = Trim(CStr(tempVal))
                
                If key <> "" And LCase(key) <> "total" Then
                    currentGSTIN = ""
                    currentPayType = ""
                    If dictGSTIN.exists(key) Then currentGSTIN = dictGSTIN(key)
                    If dictPayType.exists(key) Then currentPayType = dictPayType(key)
                    
                    If LCase(currentPayType) = "cash" Then
                        Set chosenWs = wsCash
                        outRow = outCash
                        outCash = outCash + 1
                    ElseIf currentGSTIN <> "" And Len(currentGSTIN) >= 10 Then
                        Set chosenWs = wsB2B
                        outRow = outB2B
                        outB2B = outB2B + 1
                    Else
                        Set chosenWs = wsB2C
                        outRow = outB2C
                        outB2C = outB2C + 1
                    End If
                    
                    If iDateCol > 0 And Not IsError(wsItems.Cells(i, iDateCol).Value) Then
                        rawDate = Trim(CStr(wsItems.Cells(i, iDateCol).Value))
                        rawDate = Replace(Replace(Replace(rawDate, Chr(10), ""), Chr(13), ""), " ", "")
                        If IsDate(rawDate) Then
                            properDate = CDate(rawDate)
                        ElseIf Len(rawDate) >= 10 And InStr(rawDate, "/") > 0 Then
                            properDate = DateSerial(Val(Right(rawDate, 4)), Val(Mid(rawDate, 4, 2)), Val(Left(rawDate, 2)))
                        Else
                            properDate = rawDate
                        End If
                        chosenWs.Cells(outRow, 1).Value = properDate
                    End If

                    If iPartyCol > 0 Then chosenWs.Cells(outRow, 2).Value = wsItems.Cells(i, iPartyCol).Text
                    chosenWs.Cells(outRow, 3).Value = currentGSTIN
                    chosenWs.Cells(outRow, 4).Value = key
                    If iItemCol > 0 Then chosenWs.Cells(outRow, 5).Value = wsItems.Cells(i, iItemCol).Text
                    If iHSNCol > 0 Then chosenWs.Cells(outRow, 6).Value = wsItems.Cells(i, iHSNCol).Text
                    If iQtyCol > 0 Then chosenWs.Cells(outRow, 7).Value = Val(wsItems.Cells(i, iQtyCol).Text)
                    If iPriceCol > 0 Then chosenWs.Cells(outRow, 8).Value = Val(wsItems.Cells(i, iPriceCol).Text)

                    ' --- SPLIT DISCOUNT ---
                    If iDiscCol > 0 Then
                        discStr = Trim(wsItems.Cells(i, iDiscCol).Text)
                        If InStr(discStr, "(") > 0 Then
                            splitData = Split(discStr, "(")
                            chosenWs.Cells(outRow, 9).Value = Val(Trim(splitData(0)))
                            chosenWs.Cells(outRow, 10).Value = Val(Replace(Replace(Trim(splitData(1)), ")", ""), "%", ""))
                        Else
                            chosenWs.Cells(outRow, 9).Value = Val(discStr)
                        End If
                    End If

                    ' --- SPLIT GST ---
                    If iGSTCol > 0 Then
                        gstStr = Trim(wsItems.Cells(i, iGSTCol).Text)
                        If InStr(gstStr, "(") > 0 Then
                            splitData = Split(gstStr, "(")
                            chosenWs.Cells(outRow, 11).Value = Val(Trim(splitData(0)))
                            chosenWs.Cells(outRow, 12).Value = Val(Replace(Replace(Trim(splitData(1)), ")", ""), "%", ""))
                        Else
                            chosenWs.Cells(outRow, 11).Value = Val(gstStr)
                        End If
                    End If

                    If iAmtCol > 0 Then chosenWs.Cells(outRow, 13).Value = Val(wsItems.Cells(i, iAmtCol).Text)
                    chosenWs.Cells(outRow, 14).Value = currentPayType
                End If
            End If
        End If
    Next i

    ' 7. ADD FORMULA TOTAL ROWS AND FORMAT OUTPUTS
    Dim finalWs As Variant, tRow As Long
    For Each finalWs In Array(wsB2B, wsB2C, wsCash)
        ' Find the last line where data was written on this specific sheet
        tRow = finalWs.Cells(finalWs.Rows.Count, "D").End(xlUp).Row
        
        ' Only inject totals if the sheet contains actual row entries
        If tRow >= 2 Then
            tRow = tRow + 2 ' Leaves 1 blank row for visual layout
            
            finalWs.Cells(tRow, 2).Value = "Total"
            finalWs.Cells(tRow, 2).Font.Bold = True
            
            ' Inject standard sum blocks
            finalWs.Cells(tRow, 7).Formula = "=SUM(G2:G" & (tRow - 2) & ")"  ' Quantity Sum
            finalWs.Cells(tRow, 9).Formula = "=SUM(I2:I" & (tRow - 2) & ")"  ' Discount Sum
            finalWs.Cells(tRow, 11).Formula = "=SUM(K2:K" & (tRow - 2) & ")" ' GST Sum
            finalWs.Cells(tRow, 13).Formula = "=SUM(M2:M" & (tRow - 2) & ")" ' Final Amount Sum
            
            ' Apply double-line accounting format border styles to the calculation block
            With finalWs.Range(finalWs.Cells(tRow, 1), finalWs.Cells(tRow, 14))
                .Font.Bold = True
                .Borders(xlEdgeTop).LineStyle = xlContinuous
                .Borders(xlEdgeBottom).LineStyle = xlDouble
            End With
        End If
        
        ' Normalize layout metrics
        finalWs.Columns(1).NumberFormat = "dd/mm/yyyy"
        finalWs.Columns("I:M").NumberFormat = "0.00"
        finalWs.Columns.AutoFit
    Next finalWs
    
    MsgBox "All sheets combined, sorted, and complete with calculation formulas!", vbInformation
End Sub
